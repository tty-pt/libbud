static component LikeButton(const char *label, const char *id) {
	int *likes = bud_use_state(int, 0);
	return (
		<button id={id} class="btn" onClick={[int *likes](bud_event *e) {
			bud_set(likes, *likes + 1);
		}}>
			{label}:{*likes}
		</button>
	);
}

component List(const char *const *songs, size_t count) {
	return (
		<>
			<TrackBadge prefix="Tracks" count={(int)count} />
			<ul>
				{for (size_t i = 0; i < count; i++)
					<li key={songs[i]}>{songs[i]}</li>
				}
				{if (count == 0) <li class="empty">None</li>}
			</ul>
		</>
	);
}

#include <bud/bud_reconcile.h>

typedef struct Track {
	const char *title;
	int bpm;
} Track;

component Picker(const Track *t, int *out) {
	int fast = t->bpm < 120;
	return <button onClick={[out, int idx = fast]() { *out = idx; }}>{t->title}</button>;
}
